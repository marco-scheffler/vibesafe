# vibesafe Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `vibesafe`, a global Claude Code skill that combines proactive secure-coding guidance with a real multi-scanner orchestrator (`scan.py`) that normalizes findings into one report.

**Architecture:** A concise `SKILL.md` brain (progressive disclosure → `references/`) plus a zero-dependency Python 3 orchestrator `scripts/scan.py`. `scan.py` detects the stack, runs each scanner (installed binary or ephemeral `npx`/`uvx`/`pipx run`, else graceful skip), normalizes heterogeneous JSON into a unified `Finding` schema, and writes `report.json` + `report.md`. Claude reads `report.json`, proposes fixes, applies them only after user approval.

**Tech Stack:** Python 3 (stdlib only: `argparse`, `subprocess`, `json`, `dataclasses`, `pathlib`, `shutil`, `tempfile`). Tests: stdlib `unittest`. External scanners: gitleaks, npm audit, pip-audit, osv-scanner, semgrep, trivy, checkov (all optional at runtime).

**Reference:** Spec at `docs/specs/2026-06-13-vibesafe-skill-design.md`.

| `SKILL.md` | Skill brain: frontmatter (`name`, `description`) + when-to-trigger, two modes, how to run scan, how to report/fix |
| `references/secure-coding.md` | Proactive guidance content (Access Control, XSS, CSRF, SSRF, Secrets, …) |
| `references/remediation.md` | Fix patterns per finding category & framework |
| `references/tools.md` | Scanner matrix: coverage, run command, install command, output format |
| `README.md`, `LICENSE`, `.gitignore` | Repo hygiene |

**Single module choice:** `scan.py` stays one file (~450 lines). It is cohesive (one orchestration pipeline) and easier to symlink/run than a package. Functions are individually pure/testable, so the single file does not hurt testability.

#!/usr/bin/env python3
"""vibesafe — multi-scanner orchestrator (stdlib only)."""
from __future__ import annotations
import argparse, json, os, shutil, subprocess, sys, tempfile, time
from dataclasses import dataclass, asdict
from pathlib import Path

SEVERITIES = ["critical", "high", "medium", "low", "info"]
_SEV_RANK = {s: i for i, s in enumerate(SEVERITIES)}
_SEV_ALIASES = {
    "moderate": "medium", "error": "high", "warning": "medium", "warn": "medium",
    "unknown": "info", "none": "info", "": "info",
}

def normalize_severity(value) -> str:
    if value is None:
        return "info"
    v = str(value).strip().lower()
    v = _SEV_ALIASES.get(v, v)
    return v if v in _SEV_RANK else "info"

def severity_sort_key(sev: str) -> int:
    return _SEV_RANK.get(normalize_severity(sev), len(SEVERITIES))

@dataclass
class Finding:
    tool: str
    category: str            # secrets|deps|sast|iac|license
    severity: str            # critical|high|medium|low|info
    title: str
    file: str | None = None
    line: int | None = None
    package: str | None = None
    cve: str | None = None
    rule_id: str | None = None
    remediation: str | None = None

def finding_to_dict(f: "Finding") -> dict:
    d = asdict(f)
    d["severity"] = normalize_severity(d["severity"])
    return d
```

def detect_stack(path: Path) -> dict:
    path = Path(path)
    def has(*globs):
        return any(next(path.rglob(g), None) is not None for g in globs)
    return {
        "node": (path / "package.json").exists() or has("package.json"),
        "python": any((path / f).exists() for f in
                      ("requirements.txt", "pyproject.toml", "Pipfile", "setup.py"))
                  or has("requirements.txt", "pyproject.toml"),
        "docker": has("Dockerfile", "*.dockerfile", "docker-compose.yml", "docker-compose.yaml"),
        "terraform": has("*.tf"),
        "git": (path / ".git").exists(),
    }
```
Note: `rglob` honors no ignore rules; acceptable for detection. Keep top-level checks first for speed.

# tools runnable via ephemeral runners
_NPX_TOOLS = set()                       # npm audit runs via `npm`, handled specially
_PY_TOOLS = {"semgrep", "checkov"}       # available via uvx / pipx run
_NATIVE_ONLY = {"gitleaks", "trivy", "osv-scanner"}  # need real install

def resolve_runner(tool: str, have=None):
    """Return a command prefix list to run `tool`, or None if unavailable."""
    have = have or (lambda t: shutil.which(t) is not None)
    if have(tool):
        return [tool]
    if tool in _PY_TOOLS:
        if have("uvx"):
            return ["uvx", tool]
        if have("pipx"):
            return ["pipx", "run", tool]
    # native-only tools: no ephemeral fallback
    return None

def run_tool(cmd, cwd, timeout=120):
    """Run a command capturing output. Returns (returncode, stdout, stderr, status)."""
    try:
        p = subprocess.run(cmd, cwd=str(cwd), capture_output=True, text=True, timeout=timeout)
        return p.returncode, p.stdout, p.stderr, "ok"
    except subprocess.TimeoutExpired:
        return None, "", f"timeout after {timeout}s", "timeout"
    except FileNotFoundError as e:
        return None, "", str(e), "missing"
    except Exception as e:  # never let one tool crash the run
        return None, "", repr(e), "errored"

INSTALL_HINTS = {
    "gitleaks": "brew install gitleaks",
    "trivy": "brew install trivy",
    "osv-scanner": "brew install osv-scanner",
    "semgrep": "brew install semgrep  # or: uvx semgrep / pipx run semgrep",
}

def _first_cve(aliases, fallback=None):
    for a in aliases or []:
        if str(a).upper().startswith("CVE-"):
            return a
    return fallback

def normalize_npm_audit(raw: dict) -> list:
    out = []
    for name, v in (raw.get("vulnerabilities") or {}).items():
        sev = normalize_severity(v.get("severity"))
        cve = None; title = f"Vulnerable dependency: {name}"
        for via in v.get("via") or []:
            if isinstance(via, dict):
                title = via.get("title", title)
                cve = cve or _first_cve([via.get("url", "")]) or (
                    via.get("url") if "advisories" in str(via.get("url", "")) else None)
        out.append(Finding(tool="npm-audit", category="deps", severity=sev, title=title,
                           package=name, cve=cve,
                           remediation=f"Upgrade {name} (npm audit fix) to a non-vulnerable range."))
    return out

def normalize_pip_audit(raw) -> list:
    deps = raw.get("dependencies") if isinstance(raw, dict) else raw
    out = []
    for dep in deps or []:
        name, ver = dep.get("name"), dep.get("version")
        for vu in dep.get("vulns") or []:
            fixes = ", ".join(vu.get("fix_versions") or []) or "a patched version"
            out.append(Finding(
                tool="pip-audit", category="deps", severity="high",  # pip-audit omits CVSS; treat known CVE as high
                title=f"Vulnerable dependency: {name} {ver} ({vu.get('id')})",
                package=name, cve=_first_cve(vu.get("aliases"), vu.get("id")),
                remediation=f"Upgrade {name} to {fixes}."))
    return out

def _cvss_to_sev(score) -> str:
    try:
        s = float(score)
    except (TypeError, ValueError):
        return "high"
    return "critical" if s >= 9 else "high" if s >= 7 else "medium" if s >= 4 else "low"

def normalize_osv(raw: dict) -> list:
    out = []
    for res in raw.get("results") or []:
        src = (res.get("source") or {}).get("path")
        for pkg in res.get("packages") or []:
            p = pkg.get("package") or {}
            max_sev = None
            for g in pkg.get("groups") or []:
                max_sev = g.get("max_severity") or max_sev
            for vu in pkg.get("vulnerabilities") or []:
                out.append(Finding(
                    tool="osv-scanner", category="deps", severity=_cvss_to_sev(max_sev),
                    title=f"Vulnerable dependency: {p.get('name')}@{p.get('version')} ({vu.get('id')})",
                    file=src, package=p.get("name"),
                    cve=_first_cve(vu.get("aliases"), vu.get("id")),
                    remediation=f"Upgrade {p.get('name')} to a fixed version."))
    return out
```

def build_report(findings, target, run, skipped, errored, duration_s) -> dict:
    findings = sorted(findings, key=lambda f: severity_sort_key(f.severity))
    counts = {s: 0 for s in SEVERITIES}
    for f in findings:
        counts[normalize_severity(f.severity)] += 1
    return {
        "summary": {**counts, "total": len(findings),
                    "scanners_run": run, "scanners_skipped": skipped,
                    "scanners_errored": errored,
                    "duration_s": round(duration_s, 2), "target": str(target)},
        "findings": [finding_to_dict(f) for f in findings],
    }

def render_markdown(rep: dict) -> str:
    s = rep["summary"]; L = []
    L.append(f"# vibesafe report — `{s['target']}`")
    L.append("")
    L.append("| Severity | Count |")
    L.append("|---|---|")
    for sev in SEVERITIES:
        L.append(f"| {sev} | {s.get(sev,0)} |")
    L.append("")
    L.append("| Sev | Category | Finding | Location |")
    L.append("|---|---|---|---|")
    for f in rep["findings"]:
        loc = (f.get("file") or "")
        if f.get("line"): loc += f":{f['line']}"
        L.append(f"| {f['severity']} | {f['category']} | {f['title'].replace('|','/')} | {loc} |")
    L.append("")
    run = ", ".join(s["scanners_run"]) or "none"
    L.append(f"**Coverage:** ran: {run}")
    if s["scanners_skipped"]:
        sk = "; ".join(f"{x['tool']} ({x['reason']} → `{x.get('install','')}`)" for x in s["scanners_skipped"])
        L.append(f" · skipped: {sk}")
    if s["scanners_errored"]:
        L.append(" · errored: " + ", ".join(x["tool"] for x in s["scanners_errored"]))
    L.append(f" · {s['duration_s']}s")
    return "\n".join(L)
```

def _run_scanner(tool, argv, normalizer, cwd, timeout):
    """Returns (findings, status_dict). status_dict goes into run/skipped/errored."""
    runner = resolve_runner(tool)
    if runner is None:
        return [], {"_state": "skipped", "tool": tool, "reason": "not installed",
                    "install": INSTALL_HINTS.get(tool, "")}
    rc, out, err, status = run_tool(runner + argv, cwd=cwd, timeout=timeout)
    if status in ("timeout", "errored", "missing"):
        return [], {"_state": "errored", "tool": tool, "reason": status}
    try:
        data = json.loads(out) if out.strip() else ({} if tool != "gitleaks" else [])
        return normalizer(data), {"_state": "run", "tool": tool}
    except json.JSONDecodeError:
        return [], {"_state": "errored", "tool": tool, "reason": "bad json"}

def _plan(stack, only):
    """Yield (tool, argv, normalizer) for applicable scanners."""
    jobs = []
    # secrets (always, if git/source present)
    jobs.append(("gitleaks", ["detect", "--no-banner", "--no-git", "-f", "json", "-r", "/dev/stdout"], normalize_gitleaks))
    if stack["node"]:
        jobs.append(("npm", ["audit", "--json"], normalize_npm_audit))   # special-cased below
    if stack["python"]:
        jobs.append(("pip-audit", ["-f", "json"], normalize_pip_audit))
    jobs.append(("osv-scanner", ["--format", "json", "-r", "."], normalize_osv))
    jobs.append(("semgrep", ["--config", "p/security-audit", "--config", "p/secrets",
                             "--config", "p/owasp-top-ten", "--json", "--quiet", "."], normalize_semgrep))
    if stack["docker"] or stack["terraform"]:
        jobs.append(("trivy", ["fs", "--format", "json", "--scanners",
                               "vuln,misconfig,secret,license", "."], normalize_trivy))
    if only:
        cat_of = {"gitleaks":"secrets","npm":"deps","pip-audit":"deps","osv-scanner":"deps",
                  "semgrep":"sast","trivy":"iac"}
        jobs = [j for j in jobs if cat_of[j[0]] in only]
    return jobs

def main(argv=None) -> int:
    ap = argparse.ArgumentParser(prog="scan.py", description="vibesafe multi-scanner")
    ap.add_argument("path", nargs="?", default=".")
    ap.add_argument("--only", default="", help="comma list: secrets,deps,sast,iac,license")
    ap.add_argument("--out-dir", default=None)
    ap.add_argument("--timeout", type=int, default=120)
    ap.add_argument("--no-install-hints", action="store_true")
    a = ap.parse_args(argv)

    target = Path(a.path).resolve()
    out_dir = Path(a.out_dir) if a.out_dir else Path(tempfile.mkdtemp(prefix="vibesafe-"))
    out_dir.mkdir(parents=True, exist_ok=True)
    only = {x.strip() for x in a.only.split(",") if x.strip()}
    stack = detect_stack(target)

    all_findings, run, skipped, errored = [], [], [], []
    for tool, base_argv, norm in _plan(stack, only):
        real_tool = "npm" if tool == "npm" else tool
        # npm audit: special — `npm` is the binary, subcommand `audit`
        if tool == "npm":
            runner = resolve_runner("npm")
            if runner is None:
                skipped.append({"tool":"npm","reason":"not installed","install":""}); continue
            rc, o, e, st = run_tool(runner + base_argv, cwd=target, timeout=a.timeout)
            try:
                fs = norm(json.loads(o)) if o.strip() else []
                all_findings += fs; run.append("npm-audit")
            except json.JSONDecodeError:
                errored.append({"tool":"npm-audit","reason":"bad json"})
            continue
        fs, status = _run_scanner(tool, base_argv, norm, cwd=target, timeout=a.timeout)
        all_findings += fs
        bucket = {"run": run, "skipped": skipped, "errored": errored}[status.pop("_state")
                  if status["_state"] != "run" else "run"]
        # simplify: route by state
    # (Step 3 note: see cleaned routing in Step 3b)
    rep = build_report(all_findings, target, run, skipped, errored, time.time() - t0)
    (out_dir / "report.json").write_text(json.dumps(rep, indent=2))
    md = render_markdown(rep)
    (out_dir / "report.md").write_text(md)
    print(md); print(f"\n[vibesafe] report: {out_dir}/report.json")
    return 0

if __name__ == "__main__":
